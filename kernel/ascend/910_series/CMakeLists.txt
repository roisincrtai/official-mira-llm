cmake_minimum_required(VERSION 3.19)
project(mira_ascend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT SOC_VERSION MATCHES "^Ascend910[A-Za-z0-9_]*$")
    message(FATAL_ERROR "SOC_VERSION must name the exact Ascend 910-series processor")
endif()
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit")
if(NOT ASCEND_CANN_PACKAGE_PATH)
    message(FATAL_ERROR "ASCEND_HOME_PATH or ASCEND_CANN_PACKAGE_PATH is required")
endif()
if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
elseif(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake")
else()
    message(FATAL_ERROR "CANN Ascend C CMake support is missing")
endif()
include(${ASCENDC_CMAKE_DIR}/ascendc.cmake)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import torch; print(torch.utils.cmake_prefix_path)"
                OUTPUT_VARIABLE TORCH_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${TORCH_PREFIX}")
find_package(Torch REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pathlib, torch_npu; print(pathlib.Path(torch_npu.__file__).parent)"
                OUTPUT_VARIABLE TORCH_NPU_PATH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
ascendc_library(mira_device STATIC affine.cpp)
add_library(mira_ascend MODULE binding.cpp)
target_compile_definitions(mira_ascend PRIVATE MIRA_SOC_VERSION="${SOC_VERSION}")
set_target_properties(mira_ascend PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(mira_ascend PRIVATE ${TORCH_FLAGS})
target_include_directories(mira_ascend PRIVATE ${TORCH_INCLUDE_DIRS} ${Python3_INCLUDE_DIRS}
                          "${TORCH_NPU_PATH}/include" "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(mira_ascend PRIVATE "${TORCH_NPU_PATH}/lib" "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(mira_ascend PRIVATE mira_device ${TORCH_LIBRARIES} torch_python torch_npu
                      Python3::Python ascendcl)
set_target_properties(mira_ascend PROPERTIES BUILD_RPATH "${TORCH_NPU_PATH}/lib;${ASCEND_CANN_PACKAGE_PATH}/lib64")
